stage-release.sh    — раскладка tar-архива релиза по манифесту (инструмент упаковки)
check-shipping.sh   — сверка манифеста с .ipk/develop (инструмент упаковки)
egress-failover.sh  — внешний наблюдатель "интерфейс -> Xray"; штатный путь —

                      (два механизма одновременно конфликтуют)
